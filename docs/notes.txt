
# 1. Point to the new CUDA 11.8 installation
export PATH=/usr/local/cuda-11.8/bin:$PATH
export LD_LIBRARY_PATH=/usr/local/cuda-11.8/lib64:$LD_LIBRARY_PATH

# 2. Force the build system to use GCC 11 (Critical for CUDA 11.8)
export CC=/usr/bin/gcc-11
export CXX=/usr/bin/g++-11
export CUDACXX=/usr/local/cuda-11.8/bin/nvcc

-------------------------------------------------------------

rm -rf build

cmake -B build \
-DWHISPER_SDL2=ON \
-DGGML_CUDA=ON \
-DGGML_VULKAN=ON \
-DCMAKE_C_COMPILER=/usr/bin/gcc-11 \
-DCMAKE_CXX_COMPILER=/usr/bin/g++-11 \
-DCMAKE_CUDA_HOST_COMPILER=/usr/bin/g++-11


cmake --build build -j 16 --config Release

-------------------------------------------------------------

./build/bin/whisper-stream -m ./models/ggml-tiny.bin -t 8 --step 500 --length 5000 -l es

-------------------------------------------------------------

ffmpeg -i output.wav -f segment -segment_time 10 -c copy parts/out%03d.wav
